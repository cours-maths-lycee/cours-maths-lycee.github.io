\entetefiche{Chapitre 3 --- Vecteurs}{Seconde}

{\small Sur un quadrillage, on lit chaque vecteur comme un \textbf{déplacement} : tant
de carreaux à droite ou à gauche, tant vers le haut ou vers le bas. Sur les figures, le
vecteur donné est en couleur ; un représentant reporté garde sa couleur et son nom ; le
vecteur cherché est en noir.\par}

\partie{Caractéristiques d'un vecteur}

\begin{exo}[Égaux ou non]
\explique{Pour chaque égalité, on trace les deux flèches, de la première lettre vers la
seconde, puis on compare leurs déplacements.}{méthode 1 : même déplacement, vecteurs
égaux}
\cas{a)}
\cotefig{%
\explique{$\vect{KL}$ : $4$ carreaux à droite.}{de $K$ vers $L$}
\explique{$\vect{NP}$ : $4$ carreaux à droite.}{de $N$ vers $P$}
\explique{Même déplacement, donc $\vect{KL}\mathrel{\res{=}}\vect{NP}$.}{même direction, même sens,
même norme}}
{\begin{tikzpicture}[scale=0.36]\figKLM
  \draw[vecteur] (0,0)--(4,0) node[nom,above]{$\vect{KL}$};
  \draw[vecteurB] (3,3)--(7,3) node[nom,below]{$\vect{NP}$};
\end{tikzpicture}}
\cas{b)}
\cotefig{%
\explique{$\vect{KN}$ : $3$ carreaux à droite et $3$ vers le haut.}{de $K$ vers $N$}
\explique{$\vect{LP}$ : $3$ carreaux à droite et $3$ vers le haut.}{de $L$ vers $P$}
\explique{Donc $\vect{KN}\mathrel{\res{=}}\vect{LP}$.}{même déplacement}}
{\begin{tikzpicture}[scale=0.36]\figKLM
  \draw[vecteur] (0,0)--(3,3) node[nom,above left]{$\vect{KN}$};
  \draw[vecteurB] (4,0)--(7,3) node[nom,right]{$\vect{LP}$};
\end{tikzpicture}}
\cas{c)}
\cotefig{%
\explique{$\vect{KP}$ : $7$ carreaux à droite et $3$ vers le haut.}{de $K$ vers $P$}
\explique{$\vect{LR}$ : $7$ carreaux à droite et $3$ vers le haut.}{de $L$ vers $R$}
\explique{Donc $\vect{KP}\mathrel{\res{=}}\vect{LR}$.}{même déplacement}}
{\begin{tikzpicture}[scale=0.36]\figKLM
  \draw[vecteur] (0,0)--(7,3) node[nom,above left]{$\vect{KP}$};
  \draw[vecteurB] (4,0)--(11,3) node[nom,below right]{$\vect{LR}$};
\end{tikzpicture}}
\cas{d)}
\cotefig{%
\explique{$\vect{NL}$ : $1$ carreau à droite et $3$ vers le bas.}{de $N$ vers $L$}
\explique{$\vect{KN}$ : $3$ carreaux à droite et $3$ vers le haut.}{de $K$ vers $N$}
\explique{Les déplacements sont différents, donc $\vect{NL}\mathrel{\res{\neq}}\vect{KN}$.}{les
directions ne sont même pas les mêmes}}
{\begin{tikzpicture}[scale=0.36]\figKLM
  \draw[vecteur] (3,3)--(4,0) node[nom,right]{$\vect{NL}$};
  \draw[vecteurB] (0,0)--(3,3) node[nom,above left]{$\vect{KN}$};
\end{tikzpicture}}
\cas{e)}
\cotefig{%
\explique{$\vect{KM}$ : $8$ carreaux à droite.}{de $K$ vers $M$}
\explique{$\vect{NR}$ : $8$ carreaux à droite.}{de $N$ vers $R$}
\explique{Donc $\vect{KM}\mathrel{\res{=}}\vect{NR}$.}{même déplacement}}
{\begin{tikzpicture}[scale=0.36]\figKLM
  \draw[vecteur] (0,0)--(8,0) node[nom,above]{$\vect{KM}$};
  \draw[vecteurB] (3,3)--(11,3) node[nom,below]{$\vect{NR}$};
\end{tikzpicture}}
\cas{f)}
\cotefig{%
\explique{$\vect{PR}$ : $4$ carreaux à droite.}{de $P$ vers $R$}
\explique{$\vect{PN}$ : $4$ carreaux à gauche.}{de $P$ vers $N$}
\explique{Donc $\vect{PR}\mathrel{\res{\neq}}\vect{PN}$.}{les sens sont contraires}}
{\begin{tikzpicture}[scale=0.36]\figKLM
  \draw[vecteur] (7,3)--(11,3) node[nom,below]{$\vect{PR}$};
  \draw[vecteurB] (7,3)--(3,3) node[nom,below]{$\vect{PN}$};
\end{tikzpicture}}
\rem{f) Même direction, même norme, mais sens contraires : $\vect{PN}$ et $\vect{PR}$ sont
opposés, $\vect{PN}=-\vect{PR}$. Deux flèches de même longueur sur une même droite ne
suffisent pas : il faut regarder où est la pointe.}
\end{exo}

\begin{exo}[Lire sur la figure]
\explique{On garde la figure de l'exercice 1 et on lit chaque déplacement.}{$K$, $L$, $M$
en bas ; $N$, $P$, $R$ en haut, $4$ carreaux entre deux points voisins}
\cas{1)}
\explique{$\vect{KN}$ fait $3$ carreaux à droite et $3$ vers le haut ; $\vect{LP}$ et
$\vect{MR}$ aussi. Donc $\vect{KN}=\res{\vect{LP}}$ (ou $\vect{MR}$).}{les côtés obliques
des parallélogrammes}
\explique{$\vect{LM}$ fait $4$ carreaux à droite ; $\vect{KL}$, $\vect{NP}$ et $\vect{PR}$
aussi. Donc $\vect{LM}=\res{\vect{KL}}$ (ou $\vect{NP}$, $\vect{PR}$).}{un seul exemple
suffit}
\cas{2)}
\explique{Un vecteur opposé à $\vect{KN}$ fait $3$ carreaux à gauche et $3$ vers le bas :
$\res{\vect{NK}}$, ou $\vect{PL}$, ou $\vect{RM}$.}{on échange les deux lettres : le sens
change}
\explique{Un vecteur opposé à $\vect{KL}$ fait $4$ carreaux à gauche : $\res{\vect{LK}}$,
ou $\vect{ML}$, $\vect{PN}$, $\vect{RP}$.}{même direction, même norme, sens contraire}
\cas{3)}
\explique{Même direction et même sens, mais pas égaux : $\res{\vect{KL}}$ et
$\res{\vect{KM}}$.}{tous deux vers la droite, mais $4$ et $8$ carreaux : normes
différentes}
\explique{Même norme, mais pas égaux : $\res{\vect{KL}}$ et $\res{\vect{PN}}$.}{tous deux
de $4$ carreaux, mais de sens contraires}
\explique{Même direction et sens contraires, mais pas opposés : $\res{\vect{KM}}$ et
$\res{\vect{LK}}$.}{$8$ carreaux à droite, $4$ à gauche : normes différentes}
\rem{Deux vecteurs égaux ont les trois caractéristiques en commun ; il suffit qu'une seule
diffère pour qu'ils ne le soient plus. Ici, d'autres réponses conviennent.}
\end{exo}

\begin{exo}[Construire un représentant]
\explique{On trace d'abord $\vec{u}$ : une flèche de $3$~cm. Puis on construit chaque
$\vec{v}$ en partant de ses trois caractéristiques.}{figures réduites de moitié}
\cas{1)}
\cotefig{%
\explique{On trace une flèche parallèle à $\vec{u}$, pointée du même côté, et deux fois
plus longue : $6$~cm.}{même direction, même sens, norme $6$~cm}}
{\begin{tikzpicture}[scale=0.5]
  \draw[vecteur] (0,1)--(3,1) node[nom,above]{$\vec{u}$};

\begin{exo}[Dans un parallélogramme]
\vspace{0.55\baselineskip}\cotefig[0.28]{%
\explique{Dans le parallélogramme $RSTU$, les côtés opposés $[RS]$ et $[UT]$, $[RU]$ et
$[ST]$ sont parallèles et de même longueur.}{on fait le tour $R$, $S$, $T$, $U$}
\explique{Ses diagonales $[RT]$ et $[SU]$ se coupent en leur milieu $O$.}{propriété du
collège}}
{\begin{tikzpicture}[scale=0.5]
  \draw[fig] (0,0)--(4,0)--(5,2)--(1,2)--cycle;
  \draw[aide] (0,0)--(5,2) (4,0)--(1,2);
  \pt{0,0}{below left}{R} \pt{4,0}{below right}{S} \pt{5,2}{above right}{T}
  \pt{1,2}{above left}{U} \pt{2.5,1}{below}{O}
\end{tikzpicture}}
\cas{1)}
\explique{$\vect{RS}=\res{\vect{UT}}$.}{côté opposé, parcouru dans le même sens}
\explique{$\vect{TU}=\res{\vect{SR}}$.}{de $T$ vers $U$ : on va vers la gauche, comme de $S$
vers $R$}
\explique{$\vect{RU}=\res{\vect{ST}}$.}{les deux autres côtés opposés}
\cas{2)}
\explique{$O$ est le milieu de $[RT]$ : de $R$ à $O$, puis de $O$ à $T$, on fait le même

\explique{Et $\vect{SO}=\vect{O\res{U}}$.}{}

\end{exo}

\begin{exo}[Lire des déplacements]
\explique{Pour chaque vecteur, on part de l'origine et on compte les carreaux jusqu'à la
pointe : d'abord horizontalement, puis verticalement.}{méthode 2}
\explique{$\vec{a}$ : \res{$3$ carreaux à droite et $4$ vers le haut}.}{la pointe est en

\explique{$\vec{b}$ : \res{$4$ carreaux à droite et $1$ vers le bas}.}{}

gauche}

\explique{$\vec{e}$ : \res{$3$ carreaux à gauche et $3$ vers le haut}.}{}

\partie{Translation}

\begin{exo}[Image d'un quadrilatère]
\vspace{0.55\baselineskip}\cotefig[0.4]{%
\explique{$\vec{u}$ : $5$ carreaux à droite et $1$ vers le haut.}{on lit le déplacement}
\explique{On le reporte depuis $A$, $B$, $C$ et $D$ : on obtient $A'$, $B'$, $C'$ et
$D'$.}{méthode 3 : chaque sommet glisse de la même façon}
\explique{On relie $A'$, $B'$, $C'$, $D'$ dans cet ordre.}{dans le même ordre que $A$, $B$,
$C$, $D$}}
{\begin{tikzpicture}[scale=0.42]
  \quadrillage{-0.4}{-0.4}{9.4}{7.4}
  \draw[vecteur] (3,6)--(8,7) node[nom,above]{$\vec{u}$};
  \draw[fig] (0,1)--(3,0)--(4,3)--(1,4)--cycle;
  \draw[figC] (5,2)--(8,1)--(9,4)--(6,5)--cycle;
  \draw[vecteur,line width=0.6pt] (0,1)--(5,2) node[nom,above,pos=0.4]{$\vec{u}$};
  \draw[vecteur,line width=0.6pt] (3,0)--(8,1) node[nom,below,pos=0.6]{$\vec{u}$};
  \draw[vecteur,line width=0.6pt] (4,3)--(9,4) node[nom,below,pos=0.6]{$\vec{u}$};
  \draw[vecteur,line width=0.6pt] (1,4)--(6,5) node[nom,above,pos=0.4]{$\vec{u}$};
  \pt{0,1}{left}{A} \pt{3,0}{below}{B} \pt{4,3}{left}{C} \pt{1,4}{above left}{D}
  \pt{5,2}{below}{A'} \pt{8,1}{below}{B'} \pt{9,4}{right}{C'} \pt{6,5}{above}{D'}
\end{tikzpicture}}
\rem{Le quadrilatère glisse sans tourner ni changer de forme : $A'B'C'D'$ est superposable
à $ABCD$ (propriété 1).}
\end{exo}

\begin{exo}[Dire avec les mots]
\explique{$\vec{u}$ : $3$ carreaux à droite et $3$ vers le haut ; $\vec{v}$ : $4$ à droite
et $1$ vers le bas.}{on lit les déplacements}
\begin{center}
\begin{tikzpicture}[scale=0.4]
  \quadrillage{-0.4}{-0.4}{12.4}{5.4}
  \draw[vecteur]  (0,2)--(3,5) node[nom,above left]{$\vec{u}$};
  \draw[vecteurB] (5,5)--(9,4) node[nom,above]{$\vec{v}$};
  \draw[vecteur]  (1,0)--(4,3) node[nom,above left]{$\vec{u}$};
  \draw[vecteurB] (5,2)--(9,1) node[nom,below]{$\vec{v}$};
  \draw[vecteur]  (7,0)--(10,3) node[nom,above left]{$\vec{u}$};
  \pt{1,0}{below}{A} \pt{4,3}{above}{B} \pt{5,2}{left}{C} \pt{9,1}{right}{D}
  \pt{7,0}{below}{E} \pt{10,3}{above right}{F}
\end{tikzpicture}
\end{center}
\cas{1)}
\explique{Depuis $A$, $3$ à droite et $3$ vers le haut : c'est $B$.}{$A$ est l'origine de
$\vect{AB}$}
\explique{\res{$B$ est l'image de $A$ par la translation de vecteur $\vec{u}$}.}{$A$
glisse jusqu'en $B$}
\cas{2)}
\explique{Depuis $C$, $4$ à droite et $1$ vers le bas : c'est $D$.}{$C$ est l'origine de
$\vect{CD}$}
\explique{\res{$D$ est l'image de $C$ par la translation de vecteur $\vec{v}$}.}{le point
de départ, puis le vecteur}
\cas{3)}
\explique{Depuis $E$, $3$ à droite et $3$ vers le haut : c'est $F$.}{on reporte $\vec{u}$
depuis $E$}
\explique{L'égalité : \res{$\vect{EF}=\vec{u}$}.}{le vecteur va du point à son image}
\cas{4)}
\explique{Pour aller de $D$ à $C$, on fait le déplacement de $\vec{v}$ à l'envers : c'est
le vecteur $\vect{DC}$, opposé à $\vec{v}$.}{$\vect{DC}=-\vec{v}$}
\explique{\res{$C$ est l'image de $D$ par la translation de vecteur $\vect{DC}$}, autrement
dit de vecteur $-\vec{v}$.}{le vecteur va du point de départ à son image}
\rem{La phrase suit toujours le même ordre : « [arrivée] est l'image de [départ] par la
translation de vecteur [vecteur] ». Elle se traduit par une égalité dont le premier
vecteur va du départ à l'arrivée : $\vect{AB}=\vec{u}$. Erreur fréquente en 4) : répondre $\vec{v}$, qui
transforme $C$ en $D$, et non $D$ en $C$.}
\end{exo}

\begin{exo}[Origine ou extrémité]

\partie{Relation de Chasles}

\begin{exo}[Un seul vecteur, si c'est possible]
\explique{La relation de Chasles s'applique quand la lettre d'arrivée du premier vecteur
est la lettre de départ du second ; on peut d'abord échanger l'ordre.}{théorème 2 :
$\vect{AB}+\vect{BC}=\vect{AC}$}
\cas{a)}
\begin{calculs}
\vect{BE}+\vect{EG} &= \res{\vect{BG}} &&\justif{déjà rangés : $B\to E\to G$}
\end{calculs}
\cas{b)}
\explique{$\vect{DA}+\vect{DC}$ : \res{impossible}.}{les deux vecteurs partent de $D$ :
aucune lettre ne relie l'arrivée de l'un au départ de l'autre}
\cas{c)}
\begin{calculs}
\vect{TU}+\vect{KT} &= \vect{KT}+\vect{TU} &&\justif{on range : $K\to T\to U$}\\
\vect{TU}+\vect{KT} &= \res{\vect{KU}} &&\justif{Chasles}
\end{calculs}
\cas{d)}
\begin{calculs}
\vect{FH}+\vect{HS} &= \res{\vect{FS}} &&\justif{déjà rangés : $F\to H\to S$}
\end{calculs}
\cas{e)}
\begin{calculs}
\vect{MN}+\vect{NM} &= \vect{MM} &&\justif{aller de $M$ à $N$, puis revenir en $M$}\\
\vect{MN}+\vect{NM} &= \res{\vec{0}} &&\justif{on est revenu au point de départ}
\end{calculs}
\cas{f)}
\explique{$\vect{AB}+\vect{CE}$ : \res{impossible}.}{l'arrivée $B$ n'est pas le départ $C$,
et $E$ n'est pas $A$}
\cas{g)}
\begin{calculs}
\vect{AB}-\vect{CB} &= \vect{AB}+\vect{BC} &&\justif{$-\vect{CB}=\vect{BC}$ : on inverse les lettres}\\
\vect{AB}-\vect{CB} &= \res{\vect{AC}} &&\justif{Chasles : $A\to B\to C$}
\end{calculs}
\cas{h)}
\begin{calculs}
\vect{KL}-\vect{KM} &= \vect{KL}+\vect{MK} &&\justif{$-\vect{KM}=\vect{MK}$}\\
\vect{KL}-\vect{KM} &= \vect{MK}+\vect{KL} &&\justif{on range : $M\to K\to L$}\\
\vect{KL}-\vect{KM} &= \res{\vect{ML}} &&\justif{Chasles}
\end{calculs}
\cas{i)}
\begin{calculs}
-\vect{RS}+\vect{RT} &= \vect{SR}+\vect{RT} &&\justif{$-\vect{RS}=\vect{SR}$}\\
-\vect{RS}+\vect{RT} &= \res{\vect{ST}} &&\justif{Chasles : $S\to R\to T$}
\end{calculs}
\cas{j)}
\begin{calculs}
\vect{AB}-\vect{BC} &= \vect{AB}+\vect{CB} &&\justif{$-\vect{BC}=\vect{CB}$}
\end{calculs}
\explique{$\vect{AB}+\vect{CB}$ : \res{impossible}.}{les deux vecteurs arrivent en $B$ : ni
$B$ n'est $C$, ni $B$ n'est $A$}
\rem{En b), la somme existe bien (méthode 5), mais elle ne s'écrit pas avec les lettres de
l'énoncé : il faudrait construire un nouveau point. Avec un signe moins (g à j), on change
d'abord le signe en inversant les lettres, puis on cherche le chemin ; en j), le signe
moins semblait faire un chemin $A\to B\to C$, mais il retourne le second vecteur.}
\end{exo}

\begin{exo}[Sans signe moins]
\explique{Le vecteur opposé a la même direction, la même longueur et le sens contraire :
on échange l'origine et l'extrémité.}{théorème 2 : $-\vect{AB}=\vect{BA}$}
\cas{a)}
\begin{calculs}
-\vect{AB} &= \res{\vect{BA}} &&\justif{on inverse les lettres}
\end{calculs}
\cas{b)}
\begin{calculs}
-\vect{MN} &= \res{\vect{NM}} &&\justif{on inverse les lettres}
\end{calculs}
\cas{c)}
\begin{calculs}
-2\vect{CD} &= 2\times\bigl(-\vect{CD}\bigr) &&\justif{le signe moins porte sur le vecteur}\\
-2\vect{CD} &= \res{2\vect{DC}} &&\justif{$-\vect{CD}=\vect{DC}$ ; le $2$ reste}
\end{calculs}
\cas{d)}
\begin{calculs}
\vect{EF}-\vect{GH} &= \res{\vect{EF}+\vect{HG}} &&\justif{$-\vect{GH}=\vect{HG}$ ; $\vect{EF}$ ne change pas}
\end{calculs}
\cas{e)}
\begin{calculs}
3\vect{KL}-\vect{JK} &= \res{3\vect{KL}+\vect{KJ}} &&\justif{$-\vect{JK}=\vect{KJ}$}
\end{calculs}
\rem{Seul le vecteur précédé du signe moins change. En c), le coefficient garde sa valeur :
$-2\vect{CD}$ et $2\vect{DC}$ ont la même longueur, le double de $CD$, et le même sens, de $D$
vers $C$. En d) et e), Chasles ne s'applique pas : aucune lettre ne fait le lien.}
\end{exo}

\begin{exo}[Simplifier]
\explique{On remplace chaque $-\vect{XY}$ par $\vect{YX}$, puis on range les vecteurs pour
que l'extrémité de chacun soit l'origine du suivant.}{méthode 7}
\cas{$\vec{u}$}
\begin{calculs}
\vec{u} &= \vect{BC}+\vect{CA}+\vect{AB} &&\justif{déjà rangés : $B\to C\to A\to B$}\\
\vec{u} &= \vect{BB} &&\justif{on revient au point de départ}\\
\vec{u} &= \res{\vec{0}} &&\justif{$\vect{BB}=\vec{0}$}
\end{calculs}
\cas{$\vec{v}$}
\begin{calculs}
\vec{v} &= \vect{CA}-\vect{CB}+\vect{AC} &&\justif{on recopie}\\
\vec{v} &= \vect{CA}+\vect{BC}+\vect{AC} &&\justif{$-\vect{CB}=\vect{BC}$}\\
\vec{v} &= \vect{BC}+\vect{CA}+\vect{AC} &&\justif{on range : $B\to C\to A\to C$}\\
\vec{v} &= \vect{BA}+\vect{AC} &&\justif{Chasles : $\vect{BC}+\vect{CA}=\vect{BA}$}\\
\vec{v} &= \res{\vect{BC}} &&\justif{Chasles}
\end{calculs}
\cas{$\vec{w}$}
\begin{calculs}
\vec{w} &= \vect{MC}-\vect{MA}+\vect{CB} &&\justif{on recopie}\\
\vec{w} &= \vect{MC}+\vect{AM}+\vect{CB} &&\justif{$-\vect{MA}=\vect{AM}$}\\
\vec{w} &= \vect{AM}+\vect{MC}+\vect{CB} &&\justif{on range : $A\to M\to C\to B$}\\
\vec{w} &= \res{\vect{AB}} &&\justif{Chasles, deux fois}
\end{calculs}
\cas{$\vec{s}$}
\begin{calculs}
\vec{s} &= \vect{CA}+\vect{AB}-\vect{BC} &&\justif{on recopie}\\
\vec{s} &= \vect{CA}+\vect{AB}+\vect{CB} &&\justif{$-\vect{BC}=\vect{CB}$}\\
\vec{s} &= \vect{CB}+\vect{CB} &&\justif{Chasles : $\vect{CA}+\vect{AB}=\vect{CB}$}\\
\vec{s} &= \res{2\vect{CB}} &&\justif{on regroupe : $\vect{CB}$ deux fois, comme $a+a=2a$}
\end{calculs}
\cas{$\vec{t}$}
\begin{calculs}
\vec{t} &= \vect{MK}-\vect{MJ}+2\vect{KC} &&\justif{on recopie}\\
\vec{t} &= \vect{MK}+\vect{JM}+2\vect{KC} &&\justif{$-\vect{MJ}=\vect{JM}$}\\
\vec{t} &= \vect{JM}+\vect{MK}+2\vect{KC} &&\justif{on range : $J\to M\to K$}\\
\vec{t} &= \res{\vect{JK}+2\vect{KC}} &&\justif{Chasles sur les deux premiers}
\end{calculs}
\rem{Le point $M$ disparaît de $\vec{w}$ : on passe par $M$ pour aller de $A$ à $C$. Erreur
fréquente : garder $-\vect{MA}$ et tenter Chasles avec un signe moins ; on change d'abord le
signe en inversant les lettres. En $\vec{t}$, on ne regroupe qu'une partie : $2\vect{KC}$
n'est pas un chemin de $K$ à $C$ fait une fois, Chasles ne s'applique donc pas directement
à $\vect{JK}+2\vect{KC}$. On peut aussi écrire $\vect{JK}+\vect{KC}+\vect{KC}=\vect{JC}+\vect{KC}$ :
toujours deux vecteurs, la simplification s'arrête là.}
\end{exo}

\begin{exo}[Regrouper]
\explique{Un même vecteur écrit plusieurs fois se regroupe, comme en calcul littéral ; on
change d'abord les signes moins en inversant les lettres, puis on applique Chasles s'il
le faut.}{méthode 7 : $\vect{AB}+\vect{AB}=2\vect{AB}$}
\cas{a)}
\begin{calculs}
\vect{AB}+\vect{AB}+\vect{AB} &= \res{3\vect{AB}} &&\justif{comme $a+a+a=3a$}
\end{calculs}
\cas{b)}
\begin{calculs}
\vect{EF}+2\vect{EF} &= \res{3\vect{EF}} &&\justif{comme $a+2a=3a$}
\end{calculs}
\cas{c)}
\begin{calculs}
\vect{RS}-\vect{SR} &= \vect{RS}+\vect{RS} &&\justif{$-\vect{SR}=\vect{RS}$}\\
\vect{RS}-\vect{SR} &= \res{2\vect{RS}} &&\justif{on regroupe}
\end{calculs}
\cas{d)}
\begin{calculs}
\vect{DE}+\vect{EF}+\vect{DF} &= \vect{DF}+\vect{DF} &&\justif{Chasles : $\vect{DE}+\vect{EF}=\vect{DF}$}\\
\vect{DE}+\vect{EF}+\vect{DF} &= \res{2\vect{DF}} &&\justif{on regroupe}
\end{calculs}
\cas{e)}
\begin{calculs}
\vect{GH}-\vect{KH}+\vect{GK} &= \vect{GH}+\vect{HK}+\vect{GK} &&\justif{$-\vect{KH}=\vect{HK}$}\\
\vect{GH}-\vect{KH}+\vect{GK} &= \vect{GK}+\vect{GK} &&\justif{Chasles : $\vect{GH}+\vect{HK}=\vect{GK}$}\\
\vect{GH}-\vect{KH}+\vect{GK} &= \res{2\vect{GK}} &&\justif{on regroupe}
\end{calculs}
\cas{f)}
\begin{calculs}
3\vect{AB}-\vect{BA} &= 3\vect{AB}+\vect{AB} &&\justif{$-\vect{BA}=\vect{AB}$}\\
3\vect{AB}-\vect{BA} &= \res{4\vect{AB}} &&\justif{comme $3a+a=4a$}
\end{calculs}
\rem{Erreur fréquente en c) : écrire $\vect{RS}-\vect{SR}=\vec{0}$. Ce n'est pas le même
vecteur qu'on retranche : $\vect{SR}$ est l'opposé de $\vect{RS}$, et le retrancher revient à
ajouter $\vect{RS}$.}
\end{exo}

\begin{exo}[Compléter]
\explique{On cherche la lettre qui fait un chemin : l'arrivée d'un vecteur est le départ du
suivant.}{Chasles : $\vect{AB}+\vect{BC}=\vect{AC}$}
\cas{1)}
\explique{$\vect{BC}=\vect{BA}+\vect{A\res{C}}$.}{de $B$ à $A$, puis de $A$ à $C$ : on va de
$B$ à $C$}
\cas{2)}
\explique{$\vect{CB}+\vect{\res{B}A}=\vect{CA}$.}{le second vecteur part de $B$, où
arrive le premier}
\cas{3)}
\explique{$\vect{A\res{B}}+\vect{CA}=\vect{CB}$.}{rangé : $\vect{CA}+\vect{AB}=\vect{CB}$}
\cas{4)}
\begin{calculs}
\vect{BC}+\vect{DA}+\vect{CD} &= \vect{BC}+\vect{CD}+\vect{DA} &&\justif{on range : $B\to C\to D\to A$}\\
\vect{BC}+\vect{DA}+\vect{CD} &= \res{\vect{BA}} &&\justif{Chasles}
\end{calculs}
\rem{En 3), on commence par ranger : le vecteur $\vect{CA}$ part de $C$, comme le résultat
$\vect{CB}$ ; il vient donc en premier.}
\end{exo}

\begin{exo}[Une égalité à démontrer]
\explique{On part du membre de gauche et on le transforme jusqu'à obtenir celui de
droite.}{on ne part pas du résultat}
\begin{calculs}
\vect{OB}-\vect{OA}+\vect{BC} &= \vect{OB}+\vect{AO}+\vect{BC} &&\justif{$-\vect{OA}=\vect{AO}$}\\
\vect{OB}-\vect{OA}+\vect{BC} &= \vect{AO}+\vect{OB}+\vect{BC} &&\justif{on range : $A\to O\to B\to C$}\\
\vect{OB}-\vect{OA}+\vect{BC} &= \vect{AC} &&\justif{Chasles, deux fois}
\end{calculs}
\explique{L'égalité \res{$\vect{OB}-\vect{OA}+\vect{BC}=\vect{AC}$} est démontrée, pour tous
points $O$, $A$, $B$, $C$.}{aucune figure particulière n'a servi}
\end{exo}

\begin{exo}[Exprimer avec deux vecteurs]
\explique{On fait apparaître les vecteurs demandés avec Chasles (on passe par le point
utile), puis on calcule comme en calcul littéral.}{$k(\vec{u}+\vec{v})=k\vec{u}+k\vec{v}$}
\cas{1) a)}
\begin{calculs}
\vec{u} &= \vect{CB} &&\justif{on recopie}\\
\vec{u} &= \vect{CA}+\vect{AB} &&\justif{Chasles : on passe par $A$}\\
\vec{u} &= -\vect{AC}+\vect{AB} &&\justif{$\vect{CA}=-\vect{AC}$}\\
\vec{u} &= \res{\vect{AB}-\vect{AC}} &&\justif{on range}
\end{calculs}
\cas{1) b)}
\begin{calculs}
\vec{u} &= 3\vect{CB}+\vect{BA} &&\justif{on recopie}\\
\vec{u} &= 3\bigl(\vect{AB}-\vect{AC}\bigr)-\vect{AB} &&\justif{d'après a) ; $\vect{BA}=-\vect{AB}$}\\
\vec{u} &= 3\vect{AB}-3\vect{AC}-\vect{AB} &&\justif{on développe}\\
\vec{u} &= \res{2\vect{AB}-3\vect{AC}} &&\justif{on réduit : $3-1=2$}
\end{calculs}
\cas{2) a)}
\begin{calculs}
\vec{v} &= \vect{BA}+\vect{CA} &&\justif{on recopie}\\
\vec{v} &= \vect{BC}+\vect{CA}+\vect{CA} &&\justif{Chasles : $\vect{BA}=\vect{BC}+\vect{CA}$}\\
\vec{v} &= \res{\vect{BC}+2\vect{CA}} &&\justif{on réduit}
\end{calculs}
\cas{2) b)}
\begin{calculs}
\vec{v} &= \vect{AB}+2\vect{BC}+\vect{AC} &&\justif{on recopie}\\
\vec{v} &= \vect{AB}+\vect{BC}+\vect{BC}+\vect{AC} &&\justif{$2\vect{BC}=\vect{BC}+\vect{BC}$}\\
\vec{v} &= \vect{AC}+\vect{BC}+\vect{AC} &&\justif{Chasles : $\vect{AB}+\vect{BC}=\vect{AC}$}\\
\vec{v} &= \vect{BC}+2\vect{AC} &&\justif{on réduit}\\

1), tout part de $A$ ; en 2), on passe par $C$. On peut vérifier 1) b) sur une figure en
construisant les deux membres depuis un même point.}
\end{exo}
