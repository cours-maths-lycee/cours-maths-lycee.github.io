\partie{Suites arithmétiques}

\begin{exo}[Reconnaître une suite arithmétique]
On calcule l'écart entre deux termes qui se suivent.
\begin{questions}[label=\alph*)]
  \item Écart toujours $4$ : arithmétique, $\res{r=4}$.
  \item Écart toujours $-6$ : arithmétique, $\res{r=-6}$.
  \item Écarts $1$, puis $2$ : $\res{\text{pas arithmétique}}$.
  \item Écart toujours $0{,}5$ : arithmétique, $\res{r=0{,}5}$.
\end{questions}
\end{exo}

\begin{exo}[Traduire une situation]
\begin{questions}[label=\alph*)]
  \item $u_{0}=4$ et $u_{n+1}=u_{n}+1{,}5$ : $\res{r=1{,}5}$.
  \item $u_{0}=18$ et $u_{n+1}=u_{n}-1{,}2$ : $\res{r=-1{,}2}$.
  \item $u_{0}=300$ et $u_{n+1}=u_{n}+25$ : $\res{r=25}$.
\end{questions}
\end{exo}

\begin{exo}[Calculer les premiers termes]
\begin{questions}[label=\alph*)]
  \item $7$ ; $10$ ; $13$ ; $\res{16}$.
  \item $7{,}5$ ; $5$ ; $2{,}5$ ; $\res{0}$.
  \item $w_{2}=-1$ ; $w_{3}=5$ ; $w_{4}=11$ ; $\res{w_{5}=17}$.
\end{questions}
\end{exo}

\begin{exo}[Trois termes à compléter]
Les deux écarts doivent être égaux :
\begin{calculs}
(2x+1)-(x-1) &= (4x-3)-(2x+1) &&\justif{même écart}\\
x+2 &= 2x-4\\
x &= 6
\end{calculs}
Les termes sont $\res{5}$ ; $\res{13}$ ; $\res{21}$, de raison $\res{r=8}$.
\end{exo}

\partie{Terme général d'une suite arithmétique}

\begin{exo}[Exprimer $u_{n}$ en fonction de $n$]
\begin{questions}[label=\alph*)]
  \item $u_{n}=u_{0}+nr=\res{5+4n}$ ; $u_{30}=5+120=\res{125}$.
  \item $v_{n}=v_{1}+(n-1)r$, soit $v_{n}=12-3(n-1)=\res{15-3n}$ ; $v_{40}=\res{-105}$.
  \item $w_{n}=w_{5}+(n-5)r$, soit $w_{n}=20+2(n-5)=\res{2n+10}$ ; $w_{25}=\res{60}$.
\end{questions}
\end{exo}

\begin{exo}[Raison à partir de deux termes]
\begin{questions}[label=\alph*)]
  \item \begin{calculs}
u_{8} &= u_{3}+5r &&\justif{$5$ sauts de $r$}\\
39 &= 14+5r\\
r &= 5
\end{calculs}
        $u_{0}=u_{3}-3r=-1$, donc $\res{u_{n}=5n-1}$.
  \item $v_{10}=v_{2}+8r$ donne $6=30+8r$, soit $r=-3$ ; $v_{0}=30+6=36$ :
        $\res{v_{n}=36-3n}$.
  \item $w_{30}=w_{10}+20r$ donne $12=7+20r$, soit $r=0{,}25$ ; $w_{0}=7-2{,}5=4{,}5$ :
        $\res{w_{n}=4{,}5+0{,}25n}$.
\end{questions}
\end{exo}

\begin{exo}[Est-ce un terme de la suite ?]
\begin{questions}[label=\arabic*.]
  \item $\res{u_{n}=8+3n}$.
  \item $8+3n=2\,024$ donne $n=672$ : $\res{2\,024=u_{672}}$. $8+3n=2\,025$ donne
        $3n=2\,017$, non divisible par $3$ : $\res{2\,025 \text{ n'est pas un terme}}$.
  \item $8+3n>1\,000$ donne $n>330{,}7$ : le premier est $\res{u_{331}=1\,001}$.
\end{questions}
\end{exo}

\begin{exo}[Une cotisation]
\begin{questions}[label=\arabic*.]
  \item $\res{a_{n}=25+1{,}5n}$ ; en 2036, $a_{10}=\res{40 \text{ €}}$.
  \item $25+1{,}5n>50$ donne $n>16{,}7$, soit $n=17$ : $\res{\text{en } 2043}$.
\end{questions}
\end{exo}

\partie{Montrer qu'une suite est (ou n'est pas) arithmétique}

\begin{exo}[Avec le terme général]
\begin{questions}[label=\alph*)]
  \item $u_{n+1}-u_{n}=5$ : arithmétique, $\res{r=5}$ et $u_{0}=2$.
  \item $v_{n+1}-v_{n}=2n+1$ dépend de $n$ : $\res{\text{pas arithmétique}}$.
  \item $w_{n}=n^{2}+8n+16-n^{2}=8n+16$ : arithmétique, $\res{r=8}$ et $w_{0}=16$.
  \item $t_{n}=\frac{3}{4}-\frac{1}{2}n$ : arithmétique, $\res{r=-0{,}5}$ et
        $t_{0}=0{,}75$.
\end{questions}
\end{exo}

\begin{exo}[Avec une relation de récurrence]
\begin{questions}[label=\alph*)]
  \item $u_{n+1}-u_{n}=7$ : $\res{\text{arithmétique de raison } 7}$.
  \item $1$ ; $3$ ; $7$ : écarts $2$ puis $4$ : $\res{\text{pas arithmétique}}$.
  \item $3$ ; $3$ ; $4$ : écarts $0$ puis $1$ : $\res{\text{pas arithmétique}}$.
  \item $t_{n}=2n+6-n=n+6$ : $\res{\text{arithmétique de raison } 1}$.
\end{questions}
\end{exo}

\begin{exo}[Des carrés d'allumettes]
\begin{questions}[label=\arabic*.]
  \item Chaque nouveau carré réutilise un côté : on ajoute $3$ allumettes,
        $a_{n+1}=a_{n}+3$. Avec $a_{1}=4$ : $a_{n}=4+3(n-1)=\res{3n+1}$.
  \item $a_{50}=\res{151}$.
  \item $3n+1\le 200$ donne $n\le 66{,}3$ : $\res{66 \text{ carrés}}$.
\end{questions}
\end{exo}

\begin{exo}[Somme et produit de suites arithmétiques]
\begin{questions}[label=\arabic*.]
  \item \begin{calculs}
(u_{n+1}+v_{n+1})-(u_{n}+v_{n}) &= (u_{n+1}-u_{n})+(v_{n+1}-v_{n})\\
(u_{n+1}+v_{n+1})-(u_{n}+v_{n}) &= r+r'
\end{calculs}
        $\res{(u_{n}+v_{n}) \text{ est arithmétique de raison } r+r'}$.
  \item $u_{n}v_{n}=n^{2}$ : $0$ ; $1$ ; $4$, écarts $1$ puis $3$ :
        $\res{\text{pas arithmétique}}$.
\end{questions}
\end{exo}

\partie{Variations : croissance linéaire}

\begin{exo}[Étudier et représenter]
\begin{minipage}[c]{0.58\linewidth}\raggedright
\begin{questions}[label=\arabic*.]
  \item Arithmétique de raison $1{,}5>0$ : $\res{\text{strictement croissante}}$.
  \item $2$ ; $3{,}5$ ; $5$ ; $6{,}5$ ; $8$ (graphique).
  \item $\res{y=1{,}5x+2}$ (coefficient $r$, ordonnée à l'origine $u_{0}$).
\end{questions}
\end{minipage}\hfill
\begin{minipage}[c]{0.38\linewidth}\centering
\begin{tikzpicture}[x=0.4cm,y=0.4cm]
  \reperesuite{5}{0}{9}{u_{n}}
  \gradx{1,2,3,4} \grady{2,4,6,8}
  \draw[coulCourbe,dashed] (0,2)--(4.6,8.9);
  \foreach \n in {0,1,...,4}
    \path[coulCourbe] (\n,{2+1.5*\n}) node[croix]{};
\end{tikzpicture}
\end{minipage}
\end{exo}

\begin{exo}[Sens de variation]
\begin{questions}[label=\alph*)]
  \item $r=-4<0$ : $\res{\text{strictement décroissante}}$.
  \item $r=0{,}2>0$ : $\res{\text{strictement croissante}}$.
  \item $r=0$ : $\res{\text{constante}}$.
  \item $t_{8}=t_{3}+5r$ donne $r=-3<0$ : $\res{\text{strictement décroissante}}$.
\end{questions}
\end{exo}

\begin{exo}[Lire une suite arithmétique]
\begin{questions}[label=\arabic*.]
  \item $\res{u_{0}=1}$ ; on monte de $2$ à chaque rang : $\res{r=2}$.
  \item $\res{u_{n}=1+2n}$ et $u_{20}=\res{41}$.
\end{questions}
\end{exo}

\begin{exo}[À partir de quel rang ?]
\begin{questions}[label=\alph*)]
  \item $4n>159$, $n>39{,}75$ : $\res{\text{rang } 40}$.
  \item $7n>120$ et $n>\frac{120}{7}\approx 17{,}1$ : $\res{\text{rang } 18}$.
  \item $2{,}5n\ge 59$, $n\ge 23{,}6$ : $\res{\text{rang } 24}$.
\end{questions}
\end{exo}

\partie{Somme de termes d'une suite arithmétique}

\begin{exo}[Sommes d'entiers]
\begin{questions}[label=\alph*)]
  \item $A=\frac{50\times 51}{2}=\res{1\,275}$.

\begin{exo}[Trois termes en progression géométrique]

x &= 6

Les termes sont $\res{6}$ ; $\res{12}$ ; $\res{24}$, de raison $\res{q=2}$.
